# YuE is a research repo; install it as a git checkout in the Dockerfile.
# torch / torchaudio come from the pytorch/pytorch:2.4.1-cuda12.1 base image —
# don't re-list them here, pip would re-resolve and risk pulling a mismatched
# wheel.
#
# Hard-pin the transformer stack to the combo YuE's inference/infer.py is
# written against. Floating to transformers 4.48+ breaks lazy imports
# (`Could not import module 'GenerationMixin'`); floating tokenizers /
# accelerate / huggingface_hub independently triggers similar resolution
# errors at first import.
transformers==4.45.2
tokenizers==0.20.3
accelerate==0.34.2
huggingface_hub==0.25.2
sentencepiece
einops
omegaconf
safetensors

# YuE's xcodec utils import `torch.utils.tensorboard`, which requires the
# standalone tensorboard package (absent from the pytorch base image). Without
# it inference dies at import: ModuleNotFoundError: No module named 'tensorboard'.
tensorboard
# Same story: xcodec_mini_infer/utils/utils.py does `import matplotlib` at module
# top-level, so it's needed just to load the inference code even though we never
# plot anything. Missing it dies at import: No module named 'matplotlib'.
matplotlib
# Hard-pin NumPy below 2.x. The pytorch:2.4.1 base image ships NumPy 1.x and its
# compiled extensions (torch, etc.) are built against it. Leaving matplotlib
# unpinned lets pip pull NumPy 2.x, which breaks those C-extensions at import —
# surfacing masked as transformers' "Could not import module 'GenerationMixin'".
numpy<2
